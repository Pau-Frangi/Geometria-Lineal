\section{Espacio proyectivo dual}

\begin{definición} [Espacio Vectorial Dual]
    Sea $V$ un espacio vectorial de dimensión $n+1$ sobre un cuerpo $\mathbb{K}$. El \textbf{espacio vectorial dual} de $V$, denotado por $V^*$, es el conjunto de todas las aplicaciones lineales de $V$ en $\mathbb{K}$. Es decir,
    \[
        V^* = \{ h : V \to \mathbb{K} \mid h \text{ es lineal} \}.
    \]
\end{definición}

Dada un base $\mathcal{B} = \{ \vec{u}_0, \vec{u}_1, \ldots, \vec{u}_n \}$ de $V$, existe una base dual $\mathcal{B}^* = \{ h_0, h_1, \ldots, h_n \}$ de $V^*$, donde cada $h_i$ es una forma lineal definida por:

\[
    \begin{matrix}
        h_i : V  \to  \mathbb{K} \\
        \vec{u}_j  \mapsto  \delta_{ij} =
        \begin{cases}
            1 & \text{si } i = j \\
            0 & \text{si } i \neq j
        \end{cases}
    \end{matrix}
\]

es base de $V^*$. Por lo tanto, $\dim(V^*) = n+1$. En este caso decimos que $\mathcal{B}^*$ es la \textbf{base dual} de $\mathcal{B}$.

\begin{definición} [Espacio Proyectivo Dual]
    En las condiciones anteriores, el \textbf{espacio proyectivo dual} de $\mathbb{P}=\mathbb{P}(V)$, denotado por $\mathbb{P}^*=\mathbb{P}(V^*)$, es el espacio proyectivo asociado a $V^*$. Es decir,
    \[
        \mathbb{P}^* = \mathbb{P}(V^*) = \{ [h] \mid h \in V^* \setminus \{0\} \}.
    \]
\end{definición}

Se tiene que $\dim(\mathbb{P}^*) = \dim(V^*) - 1 = \dim(\mathbb{P}) = n$.

\textbf{Interpretación:} Los puntos de $\mathbb{P}^*$ corresponden a los hiperplanos de $\mathbb{P}$. En efecto,

\[
    \underbrace{[h] \in \mathbb{P}^*}_{\text{punto}} \iff \underbrace{h \in V^* \setminus \{0\}}_{\text{recta vectorial}} \iff \underbrace{H = \{h = 0\} \subset V}_{\text{hiperplano vectorial}} \iff \underbrace{H \subset \mathbb{P}}_{\text{hiperplano proyectivo}}
\]

En coordenadas, 

\[
    \mathfrak{R} \text{ referencia en } \mathbb{P} \quad \Rightarrow \quad \mathcal{B} \text{ base de } V  
\]

\[
    \mathfrak{R}^* \text{ referencia en } \mathbb{P}^* \quad \Rightarrow \quad \mathcal{B}^* \text{ base dual de } V^*
\]

\[
    \mathbb{P}^* \ni [h] = [a_0 : a_1 : \ldots : a_n]_{\mathfrak{R}^*} \quad \Leftrightarrow \quad \mathbb{P} \supset H \equiv a_0x_0 + a_1x_1 + \ldots + a_nx_n = 0
\]
siendo $[x_0 : x_1 : \ldots : x_n]_{\mathfrak{R}}$ las coordenadas de un punto cualquiera de $\mathbb{P}$.

Dualidad entre los subespacios de $\mathbb{P}$ y $\mathbb{P}^*$:

\[
    \{ \text{Subespacios proyectivos de } \mathbb{P} \} \rightleftarrows  \{ \text{Subespacios proyectivos de } \mathbb{P}^* \}
\]
\[  
    X \to D(X) = \left\{[h] \in \mathbb{P}^* \mid X \subset H = \{h = 0\} \text{ (hiperplano proyectivo)} \right\}
\]
\[
    \Delta(X^*) = \bigcap_{[h] \in X^*} \{h = 0\} \subset \mathbb{P} \leftarrow X^* 
\]

$\Delta(X^*)$ es el \textbf{subespacio proyectivo} de $\mathbb{P}$ por ser intersección de hiperplanos proyectivos de $\mathbb{P}$.\\

$D(X)$ es \textbf{subespacio proyectivo} ya que si $X=P(W)$:\\
\[
    X \subset H \iff 
    W \subset \underbrace{\hat{H}}_{\text{hiperplano vectorial}} = \ker(h) \iff 
    h_{|W} = 0
\]

Por tanto:

\[
    X \subset H \iff
    h_{|W} = 0 \iff
    [h] \in \underbrace{Anu(W)}_{\text{anulador de } W} = \underbrace{\{ g \in V^* \mid g_{|W} = 0 \}}_{\text{subespacio vectorial de } V^*}
\]

Y decimos $D(X) = P(Anu(W))$.

\begin{lema}
    Sean $H, H_1, H_2, \ldots, H_k$ hiperplanos proyectivos de $\mathbb{P}$, entonces son equivalentes:
    \begin{enumerate}
        \item $H \supset H_1 \cap H_2 \cap \ldots \cap H_k$
        \item $H \in V \left( H_1, H_2, \ldots, H_k \right)$ en $\mathbb{P}^*$
        \item $[h] \in V \left( [h_1], [h_2], \ldots, [h_k] \right)$ en $\mathbb{P}^*$, donde $H_i = \{h_i = 0\} \quad \forall i = 1, 2, \ldots, k$
    \end{enumerate}
\end{lema}

\begin{proof}
    \[
    H \supset H_1 \cap H_2 \cap \ldots \cap H_k \text{ en } \mathbb{P} \iff
    \]
    \[
    \hat{H} \supset \hat{H}_1 \cap \hat{H}_2 \cap \ldots \cap \hat{H}_k \text{ en } V \iff
    \]
    \[
    h_1=0, h_2=0, \ldots, h_k=0 \implies h=0 \text{ en } V \iff
    \]
    \[
    h \in L \left( h_1, h_2, \ldots, h_k \right) \text{ en } V^* \iff
    \]
    \[
    [h] \in V\left( [h_1], [h_2], \ldots, [h_k] \right) \text{ en } \mathbb{P}^*
    \]
\end{proof}

Veamos la inyectividad de $\Delta$:

\begin{proof}
    Supongamos:
    \[
    \Delta(Y^*) \supset \Delta(X^*) \implies
    H \supset \Delta(X^*) \quad \forall H = \{h = 0\}: [h] \in Y^*
    \]
    Si ahora:
    \[
    X^* = V(H_0, H_1, \ldots, H_k) \underbrace{\implies}_{\text{Lema}}
    \Delta(X^*) = \bigcap_{i=0}^k H_i \implies
    \forall H \in Y^*: H \supset \Delta(X^*) = \bigcap_{i=0}^k H_i \iff
    \]
    \[
    \forall H \in Y^* \quad H \in V(H_0, H_1, \ldots, H_k) = X^* \implies
    Y^* \subset X^*
    \]
    Por tanto, finalmente:
    \[
    \Delta(Y^*) = \Delta(X^*) \implies
    \begin{cases}
        Y^* \subset X^* \\
        X^* \subset Y^*
    \end{cases} \implies
    X^* = Y^*
    \]
\end{proof}

Veamos la sobreyectividad de $D$:

\begin{proof}
    Dado $X^* = V(H_0, H_1, \ldots, H_r)$ subespacio proyectivo de $\mathbb{P}^*$, luego si definimos $X = \bigcap_{i=0}^r H_i$ tenemos que:
    \[
    D(X) =
    \{H \in P^* \mid X = \bigcap_{i=0}^r H_i \subset H \text{ en } \mathbb{P} \} \underbrace{=}_{\text{Lema}}
    V(H_0, H_1, \ldots, H_r) = X^* \text{ en } \mathbb{P}^*
    \] 
\end{proof}

Ademas, como $\Delta \circ D = \text{identidad}$, deducimos que $\Delta, D$ son biyecciones diccionario entre subespacios proyectivos del primal y dual.

\begin{observación}
    \textbf{Notación: } A partir de ahora, denotaremos $X^*=D(X)$ y diremos que $X^*$ es el subespacio proyectivo dual de $X$.\\
    Además, si $X$ es subespacio proyectivo, $X$ es intersección de hiperplanos proyectivos:
    \[
    \exists H_0, H_1, \ldots, H_k \text{ hiperplanos proyectivos tales que } X = \bigcap_{i=0}^k H_i
    \]
    Con ademas $\dim(X) = \underbrace{\dim(\mathbb{P})}_{n} - (k+1) = n - k - 1$.\\
    Entonces $X^* = V(H_0, H_1, \ldots, H_k)$ con $\dim(X^*) = k$, por tanto:
    \[
    \dim(X) + \dim(X^*) = n - k - 1 + k = n - 1
    \]
\end{observación}

\begin{proposición}[Propiedades fundamentales de la dualidad]
    Sea $\mathbb{P}$ un espacio proyectivo de dimensión $n$ y $X, Y \subset \mathbb{P}$ subespacios proyectivos. Entonces:
    \begin{enumerate}
        \item $X \subset Y \implies Y^* \subset X^*$
        \item $(X \cap Y)^* = V(X^*, Y^*)$
        \item $V(X, Y)^* = X^* \cap Y^*$
    \end{enumerate}
\end{proposición}

\textbf{Nota: } El espacio vectorial $(V^*)^*=V^{**}$ se denomina como espacio vectorial bidual de $V$, y podemos definir el siguiente isomorfismo lineal canónico de $V$ en $V^{**}$:
\[
    \begin{matrix}
        V \to  V^{**} \\
        \vec{u} \mapsto  \Theta_{\vec{u}} : V^* \to \mathbb{K} \\
        \qquad \qquad \quad h \mapsto h(\vec{u})
    \end{matrix}
\]
Este isomorfismo nos permite identificar $\mathbb{P}$ con su bidual $\mathbb{P}^{**} = \mathbb{P}(V^{**})$, ademas de cualquier subespacio proyectivo $X \subset \mathbb{P}$ con su bidual $X^{**} \subset \mathbb{P}^{**}$.

\begin{proposición}[Principio de dualidad]
    Sea $\mathbb{P}$ un espacio proyectivo de dimensión $n$, y tengamos una proposición $P$ que involucra subespacios proyectivos, contenidos, intersecciones, uniones y dimensiones. Entonces, si se obtiene otra proposición intercambiando los términos obtenemos una proposición $P'$ analoga en el espacio proyectivo dual $\mathbb{P}^*$:
    \begin{itemize}
        \item Subespacio proyectivo $X \leftrightarrow X^*$
        \item Contención $X \subset Y \leftrightarrow Y^* \subset X^*$
        \item Intersección $X \cap Y \leftrightarrow V(X^*, Y^*)$ Suma
        \item Suma $V(X, Y) \leftrightarrow X^* \cap Y^*$ Intersección
        \item Dimensión $\dim(X) = k \leftrightarrow\dim(X^*) = n - 1 - k$
    \end{itemize}
\end{proposición}

\ejemplo{
    En un espacio proyectivo $\mathbb{P}$ de dimensión $n$ tenemos que:
    \begin{itemize}
        \item Dados $2$ puntos de un plano proyectivo, $\exists !$ recta que los contiene:
        \[
        r = V(A,B)
        \]
    \end{itemize}
    Por dualidad, en el espacio proyectivo dual $\mathbb{P}^*$ de dimensión $n$ tenemos que:
    \begin{itemize}
        \item Dadas $2$ rectas de un plano proyectivo, $\exists !$ punto que pertenece a ambas:
        \[
        (r=V(A,B))^* \implies
        r^* = (V(A,B))^* =
        A^* \cap B^*
        \]
    \end{itemize}
}

Veamos que las proposiciones $P$ y $P'$ son equivalentes:

\begin{proof}
    Veamos ambas implicaciones:
    \begin{itemize}
        \item[$\implies )$] Tomamos un espacio proyectivo $\mathbb{P}$ de dimensión $n$ y una proposición $P$ cierta en él, entonces veamos que $P'$ es cierta en el espacio proyectivo dual $\mathbb{P}^*$.\\
        $P$ es cierta en cualquier espacio proyectivo de dimension $n$, luego es cierta en $\mathbb{P}^*$. Como $P$ es cierta en $\mathbb{P}^*$, $P'$ es cierta en $(\mathbb{P}^*)^* = \mathbb{P}$.
        \item[$\impliedby )$] Inmediatamente porque $(P')' = P$, y utilizamos $\implies )$ cambiando $P$ por $P'$.
    \end{itemize}
\end{proof}

Podemos volver al teorema de desargues, que nos mostraba lo siguiente:\\

Lo cual nos deja los puntos:
\begin{itemize}
    \item $D = V(B,C) \cap V(B',C')$
    \item $E = V(A,C) \cap V(A',C')$
    \item $F = V(A,B) \cap V(A',B')$
\end{itemize}

Y teniamos que probar:
\[
    D, E, F \text{ son colineales} \iff
    V(A,A') \cap V(B,B') \cap V(C,C') \neq \emptyset
\]

Y ya habiamos probado la implicación $\impliedby )$. Por dualidad, la implicación $\implies )$ es cierta:

\begin{proof}
    Dualizamos primero la proposición ya demostrada:
    \begin{itemize}
        \item[$P$ )] Si $V(A,A') \cap V(B,B') \cap V(C,C') \neq \emptyset$, entonces $D, E, F$ son colineales.
        \item[$P'$ )] Como $A,B,C,A',B',C'$ son puntos, entonces $A^* = a, B^* = b, C^* = c, A'^* = a', B'^* = b', C'^* = c'$ son rectas en el espacio proyectivo dual.\\
        Ahora tenemos que:
        \begin{itemize}
            \item $V(A,A')^* = a \cap a'$ es un punto.
            \item $V(B,B')^* = b \cap b'$ es un punto.
            \item $V(C,C')^* = c \cap c'$ es un punto.
        \end{itemize}
        Ahora podemos ver como transformar $D, E, F$ a sus duales:
        \begin{itemize}
            \item Tenemos:
            \[
            D^* = 
            (V(B,C) \cap V(B',C'))^* =
            V(V(B,C)^*, V(B',C')^*) =
            V(B^* \cap C^*, B'^* \cap C'^*) =
            V(b \cap c, b' \cap c')
            \]
            \item Analogamente obtenemos:
            \[
            E^* = V(a \cap c, a' \cap c')
            \]
            \item Y finalmente:
            \[
            F^* = V(a \cap b, a' \cap b')
            \]
        \end{itemize}
        Y la proposicion $P'$ queda:
        \[
            a \cap a', b \cap b', c \cap c' \text{ son colineales} \iff
            V(b \cap c, b' \cap c') \cap V(a \cap c, a' \cap c') \cap V(a \cap b, a' \cap b') \neq \emptyset
        \]
        Esto ultimo tambien es equivalente a que $D^*, E^*, F^*$ son concurrentes.\\
        Podemos usar la siguiente notación:
        \[
            \alpha = b \cap c, \quad \alpha' = b' \cap c', \quad
            \beta = a \cap c, \quad \beta' = a' \cap c', \quad
            \gamma = a \cap b, \quad \gamma' = a' \cap b'
        \]
        Y nos queda:
        \[
            a = V(\beta, \gamma), \quad a' = V(\beta', \gamma'), \quad
            b = V(\alpha, \gamma), \quad b' = V(\alpha', \gamma'), \quad
            c = V(\alpha, \beta), \quad c' = V(\alpha', \beta')
        \]
        La primera definicion se ve ya que $\beta \neq \gamma \iff a,b,c$ no son concurrentes.\\
        Ahora reescribiendo $P'$ en funcion de $\alpha, \beta, \gamma$ tenemos: \\
        $P'$: Dados $2$ ternas de puntos no alineados $\alpha, \beta, \gamma$ y $\alpha', \beta', \gamma'$, en un plano proyectivo, entonces si:
        \[
        V(\beta, \gamma) \cap V(\beta', \gamma'), \qquad
        V(\alpha, \gamma) \cap V(\alpha', \gamma'), \qquad
        V(\alpha, \beta) \cap V(\alpha', \beta')
        \]
        Son colineales, entonces las rectas
        \[
        V(\alpha, \alpha'), \qquad
        V(\beta, \beta'), \qquad
        V(\gamma, \gamma')
        \]
        Son concurrentes.\\
        Finalmente, como $P$ es cierta, $P'$ tambien lo es, probando asi la implicación $\implies )$ del teorema de Desargues.
    \end{itemize}
\end{proof}

\subsection{Aplicación proyectiva dual}

Recordemos que dada una aplicación lineal $\hat{f}: V \to V'$, su dual es la aplicación lineal $\hat{f}^*: V'^* \to V^*$ definida por:
\[
    \hat{f}^* : V'^* \to V^* \quad / \quad
    h \mapsto h \circ \hat{f}
\]

Se puede comprobar rapidamente la linealidad de $\hat{f}^*$ y que:
\[
\hat{f}^*(\lambda h' + \mu g') =
(\lambda h' + \mu g') \circ \hat{f} =
\lambda (h' \circ \hat{f}) + \mu (g' \circ \hat{f}) =
\lambda \hat{f}^*(h') + \mu \hat{f}^*(g')
\]

\begin{definición}[Aplicación Proyectiva Dual]
    Dada una aplicación proyectiva $f: \mathbb{P} \to \mathbb{P}'$, asociada a la aplicación lineal $\hat{f}: V \to V'$, la \textbf{aplicación proyectiva dual} de $f$ es la aplicación proyectiva $f^*: \mathbb{P}'^* \to \mathbb{P}^*$ asociada a la aplicación lineal dual $\hat{f}^*: V'^* \to V^*$.
\end{definición}

Dadas $B, B'$ bases de $V, V'$ respectivamente, y sus bases duales $B^*, B'^*$, se tiene que:

\[
    M_{B'^*, B^*}(\hat{f}^*) = (M_{B, B'}(\hat{f}))^T
\]

Y pasando a coordenadas proyectivas, con referencias $\mathfrak{R}, \mathfrak{R}'$ asociadas a las bases $B, B'$ respectivamente, y sus referencias duales $\mathfrak{R}^*, \mathfrak{R}'^*$ asociadas a las bases duales $B^*, B'^*$ respectivamente, se tiene que:

\[
    M_{\mathfrak{R}'^*, \mathfrak{R}^*}(f^*) =
    (M_{\mathfrak{R}, \mathfrak{R}'}(f))^T
\]

Entendiendo esta igualdad salvo proporcionalidad.

\subsection{Recordatorio del espacio Afín}

\begin{definición}[Espacio Afín]
    Dado $\mathbb{A} \neq \emptyset$ conjunto, se le denomina \textbf{espacio afín} si tiene asociado:
    \begin{itemize}
        \item Un espacio vectorial $\vec{\mathbb{A}}$, espacio de direcciones.
        \item Una aplicación $\varphi: \mathbb{A} \times \mathbb{A} \to \vec{\mathbb{A}}$, definida por $\varphi(A, B) = \overrightarrow{AB}$, que cumple:
        \begin{itemize}
            \item $\forall A \in \mathbb{A}$, la aplicación $\varphi_A = (A, \cdot): \mathbb{A} \to \vec{\mathbb{A}}$ definida por $B \mapsto \overrightarrow{AB}$ es biyectiva.
            \item $\forall A, B, C \in \mathbb{A}$, se cumple que $\overrightarrow{AB} + \overrightarrow{BC} = \overrightarrow{AC}$.
        \end{itemize}
    \end{itemize}
\end{definición}

\begin{corolario}
    Podemos deducir rapidamente que:
    \begin{itemize}
        \item $\forall A \in \mathbb{A}$, $\overrightarrow{AA} = \vec{0}$.
        \item $\forall A, B \in \mathbb{A}$, $\overrightarrow{AB} = -\overrightarrow{BA}$.
    \end{itemize}
\end{corolario}

Heuristicamente en un espacio afin se pueden "restar" $2$ puntos:
\[
B - A = \overrightarrow{AB}
\]

Y tambien se pueden "sumar" un punto y un vector:
\[
A + \overrightarrow{AB} = B
\]

\ejemplo{
    Todo espacio vectorial $V$ se puede ver como un espacio afín con $\vec{V} = V$ y $\varphi(u,v) : V \times V \to \vec{V} = V$ definida por $\varphi(u,v) = v - u$.
}

\begin{definición}[Referencia afín]
    Una \textbf{referencia afín} $\mathfrak{R} = \{O ; \quad P_1, P_2, \ldots, P_n\}$ son $n+1$ puntos de $\mathbb{A}$ tales que los vectores $\{\overrightarrow{OP_1}, \overrightarrow{OP_2}, \ldots, \overrightarrow{OP_n}\}$ son base de $\vec{\mathbb{A}}$.\\
\end{definición}

\textbf{Nota: } $\dim(\mathbb{A}) = \dim(\vec{\mathbb{A}}) = n$.

\begin{definición}[Coordenadas afines]
    Sea $A \in \mathbb{A}$, sus cordenadas afines respecto a la referencia afín $\mathfrak{R}$ son:
    \[
        A = (X_1, X_2, \ldots, X_n)_{\mathfrak{R}} \iff
        \overrightarrow{OA} = X_1 \overrightarrow{OP_1} + X_2 \overrightarrow{OP_2} + \ldots + X_n \overrightarrow{OP_n}
    \]
\end{definición}

\begin{proposición}[Cambio de referencia afín]
    Sean $\mathfrak{R} = \{O ; P_1, P_2, \ldots, P_n\}$ y $\mathfrak{R}' = \{O' ; P_1', P_2', \ldots, P_n'\}$ dos referencias afines de $\mathbb{A}$. Entonces:
    \[
        \begin{matrix}
            A = (X_1, X_2, \ldots, X_n)_{\mathfrak{R}} \\
            A = (X_1', X_2', \ldots, X_n')_{\mathfrak{R}'}
        \end{matrix} \implies
        \begin{pmatrix}
            1 \\ X_1' \\ X_2' \\ \vdots \\ X_n'
        \end{pmatrix} =
        \left(
        \begin{array}{c|cccc}
            1 & 0 & 0 & \ldots & 0 \\
            \hline
            q_1 & & & & \\
            q_2 & & M_{\mathcal{B}', \mathcal{B}} & & \\
            \vdots & & & & \\
            q_n & & & &
        \end{array}
        \right)
        \begin{pmatrix}
            1 \\ X_1 \\ X_2 \\ \vdots \\ X_n
        \end{pmatrix}
    \]
    Siendo $\underbrace{O}_{\text{origen de } \mathfrak{R}}= (q_1, q_2, \ldots, q_n)_{\mathfrak{R}'}$.
\end{proposición}

\begin{definición}[Subespacio afín]
    Los subespacios afines de $\mathbb{A}$ son los conjuntos de la forma $ \{ A + \vec{v} \mid \vec{v} \in \vec{B} \}$, siendo $A \in \mathbb{A}$ y $\vec{B} \subset \vec{\mathbb{A}}$ un subespacio vectorial.
\end{definición}

\begin{itemize}

    \item Un espacio afín de dimensión $n$ se puede completar a un espacio proyectivo de dimensión $n$.
\end{itemize}

    Si tenemos un plano afin $x_0 = 1$ las rectas no contenidas en $x_0 = 0$ cortan a este plano en un único punto.\\
    Podemos definir la aplicación:\\
    \[
    \pi \to \mathbb{P}^2, \quad (x_1, x_2) \mapsto (1 : x_1 : x_2)
    \]
    Todo punto de $\mathbb{P}^2$ está en la imagen de $\pi$ salvo los puntos de $\mathbb{P}(\{ x_0 = 0\})$, que es una recta proyectiva en $\mathbb{P}^2$.
    Podemos definir entonces la siguiente funcion que va del espacio afin asociado a $\mathbb{R}^2$, al cual denotaremos por $\mathbb{A}^2_{\mathbb{R}}$, al espacio proyectivo $\mathbb{P}^2_{\mathbb{R}}$ menos la recta proyectiva $\mathbb{H} = \mathbb{P}(\{ x_0 = 0\})$, asociada al hiperplano vectorial $\{ x_0 = 0\}$:
    \[
    \mathbb{A}^2_{\mathbb{R}} \to \mathbb{U} = \mathbb{P}^2_{\mathbb{R}} \setminus \mathbb{H}, \quad (x_1, x_2) \mapsto (1 : x_1 : x_2)
    \]
    Esta funcion es una biyección que "encaja" el plano afín en el plano proyectivo menos una recta proyectiva.\\
    A $\mathbb{U} = \{ x_0 \neq 0 \}$ se le llama carta afín de $\mathbb{P}^2_{\mathbb{R}}$.\\
    Ademas, es facil ver que las cartas afines $\{x_0 \neq 0\}$, $\{x_1 \neq 0\}$ y $\{x_2 \neq 0\}$ cubren todo $\mathbb{P}^2_{\mathbb{R}}$.
}

Veamos ahora la relacion entre rectas proyectivas y rectas afines:\\

Tengamos la recta afin en $\pi$ definida por la ecuación:
\[
    \{\underbrace{(1,a_1,a_2)}_{A} + t(0,v_1,v_2) \mid t \in \mathbb{R} \} \subset \pi
\]
Que en parametricas es:
\[
    \begin{cases}
        x_0 = 1 \\
        x_1 = a_1 + t v_1 \\
        x_2 = a_2 + t v_2
    \end{cases} \in \mathbb{R}^3, \quad t \in \mathbb{R}
\]
Tambien tenemos que el plano que contiene a la recta $r_A$ es el generado por los vectores $\{(1,a_1,a_2), (0,v_1,v_2)\}$, es decir:
\[
    \begin{cases}
        x_0 = s \\
        x_1 = s a_1 + t v_1 \\
        x_2 = s a_2 + t v_2
    \end{cases} \in \mathbb{R}^3, \quad s,t \in \mathbb{R}
\]
Determinando aso la recta proyectiva en $\mathbb{P}^2_{\mathbb{R}}$ definida por la ecuacion:
\[
    r_{\mathbb{P}} = \{ (s : s a_1 + t v_1 : s a_2 + t v_2) \mid s,t \in \mathbb{R}, (s,t) \neq (0,0) \} \subset \mathbb{P}^2_{\mathbb{R}}
\]
Y de hecho, podemos identificar $(s:t) \in \mathbb{P}^1_{\mathbb{R}}$

Ademas $r_{\mathbb{P}}$ contiene a la recta afin $r_A$ de partida y ademas al punto $(0:v_1:v_2)$, que corresponde al vector director de la recta afin $r_A$.\\

\]

Así, se puede escribir:
\[
\underbrace{\widetilde{\mathbb{A}}}_{\text{espacio proyectivo de dim } n} =
\underbrace{\mathbb{A}}_{\text{espacio afín de dim } n} \cup
\underbrace{\mathbb{A}_{\infty}}_{\text{hiperplano de puntos en el infinito de dim } n-1}
\]

\textbf{Construcción inversa: }


Dado un espacio vectorial \(V\) de dimensión \(n+1\), consideramos su espacio proyectivo asociado
\[
\mathbb{P} = \mathbb{P}(V),
\]
y un hiperplano proyectivo \(\mathbb{H} = \mathbb{P}(\hat{H})\), donde \(\hat{H} \subset V\) es un subespacio vectorial de dimensión \(n\).

\medskip

Entonces, el complemento \(\mathbb{P} \setminus \mathbb{H}\) admite una estructura natural de espacio afín de dimensión \(n = \dim(\mathbb{P})\).

\medskip

Tomemos en \(\mathbb{P}\) una \textbf{referencia proyectiva adaptada} al hiperplano \(\mathbb{H}\):
\[
\mathfrak{R}_{\text{proy}} = 
\left\{
\underbrace{P_0}_{\notin \mathbb{H}} = [\vec{u_0}], 
P_1 = [\vec{u_1}], \ldots, P_n = [\vec{u_n}]; 
P_{n+1}
\right\},
\]
donde los puntos \(P_1, \ldots, P_n\) pertenecen a \(\mathbb{H}\), y la base vectorial asociada es
\[
\mathcal{B}' = \{ \vec{u_0}, \underbrace{\vec{u_1}, \ldots, \vec{u_n}}_{\text{base de } \hat{H}} \}.
\]

\medskip

En coordenadas homogéneas respecto de la referencia \(\mathfrak{R}_{\text{proy}}\), el hiperplano proyectivo y su correspondiente subespacio vectorial se escriben:
\[
\mathbb{H} : x_0 = 0, 
\qquad 
\hat{H} : X_0 = 0.
\]

La carta afín asociada al complemento de \(\mathbb{H}\) es
\[
\mathbb{P} \setminus \mathbb{H} = \{ [x_0 : x_1 : \ldots : x_n] \mid x_0 \neq 0 \},
\]
entonces podemos definir la biyección que establece la correspondencia entre el espacio proyectivo $\mathbb{P} \setminus \mathbb{H}$ y el espacio afín $\vec{u_0} + \hat{H}$:
\[
\mathbb{P} \setminus \mathbb{H} \;\longrightarrow\; \vec{u_0} + \hat{H}, 
\qquad 
[\vec{v}] \;\longmapsto\; [\vec{v}] \cap (\vec{u_0} + \hat{H}).
\]

En la biyección anterior, aparece un ligero \emph{abuso de notación} que conviene aclarar.

En efecto, la intersección
\[
[\vec{v}] \cap (\vec{u_0} + \hat{H})
\]
debe entenderse como la \textbf{intersección afín} entre:
\begin{itemize}
  \item la recta afín que pasa por el origen \(O = (0,\ldots,0)\) con dirección dada por el vector \(\vec{v}\), es decir
  \[
  r_{\vec{v}} = \{\, O+t\vec{v} \mid t \in \mathbb{K} \,\},
  \]
  y
  \item el \textbf{hiperplano afín} de \(V\)
  \[
  \vec{u_0} + \hat{H} = \{\, (O+\vec{u_0}) + \vec{h} \mid \vec{h} \in \hat{H} \,\}.
  \]
\end{itemize}

Geométricamente, el conjunto \(\vec{u_0} + \hat{H}\) es un hiperplano afín de \(V\), paralelo al subespacio vectorial \(\hat{H}\), que no pasa por el origen ya que \(\vec{u_0} \notin \hat{H}\). 

Cada recta vectorial \(r_{\vec{v}}\) que pasa por el origen corta a dicho hiperplano en un único punto, lo que justifica la biyección anterior:
\[
[\vec{v}] \;\longmapsto\; r_{\vec{v}} \cap (\vec{u_0} + \hat{H}).
\]

El abuso de notación proviene de identificar un punto afín con su vector de posición respecto al origen. En rigor, el conjunto \(\vec{u_0} + \hat{H}\) debería denotarse \(A + \hat{H}\), siendo \(A=O+ \vec{u_0}\) el \emph{punto afín} cuyas coordenadas vectoriales son \(\vec{u_0}\).  
Sin embargo, cuando el origen está fijado, es habitual (aunque informal) escribir \(\vec{u_0} + \hat{H}\).

En coordenadas, la correspondencia se expresa como:
\[
(x_0 : x_1 : \ldots : x_n)_{\mathfrak{R}_{\text{proy}}}
\;\longmapsto\;
(X_1 = \tfrac{x_1}{x_0}, \, X_2 = \tfrac{x_2}{x_0}, \ldots, X_n = \tfrac{x_n}{x_0})_{\mathfrak{R}_{\text{afín}}},
\]
donde el punto de coordenadas afines \((X_1, \ldots, X_n)\) pertenece al hiperplano afín \(\vec{u_0} + \hat{H}\).

\medskip

Finalmente, la \textbf{referencia afín} asociada se define como
\[
\mathfrak{R}_{\text{afín}} = 
\{ O' = O+\vec{u_0};\; P_1 = \vec{u_1},\, P_2 = \vec{u_2},\, \ldots,\, P_n = \vec{u_n} \},
\]
cuyo conjunto de vectores directores
\[
\mathcal{B} = \{ \vec{u_1}, \vec{u_2}, \ldots, \vec{u_n} \}
\]
constituye una base del subespacio vectorial \(\hat{H}\).

Todo esto nos deja la siguiente correspondencia entre subespacios proyectivos y afines:

\[
\begin{array}{c@{\quad\leftrightarrows\quad}c@{\quad\leftrightarrows\quad}c}
    \text{Espacio Afín} & \text{Espacio Vectorial} & \text{Espacio Proyectivo} \\
    S = A + \vec{S} & \hat{S} = t\,\overrightarrow{0A} + \vec{S},\; t \in \mathbb{K} & \mathbb{P}(\hat{S}) = S \cup \underbrace{\mathbb{P}(\vec{S})}_{S_{\infty}} = \widetilde{S}
\end{array}
\]


\[
S=\widetilde{S} \setminus H = \widetilde{S} \cap (\mathbb{P} \setminus H) 
\leftarrow \qquad \leftarrow 
\widetilde{S}
\]

Siendo $\rightarrow$ la compleción, y $\leftarrow$ la restricción.\\
Obteniendo así $\widetilde{S}$, el subespacio completado con los puntos de $\infty$ de $S$.
Ademas $\widetilde{S} \cap (\mathbb{P} \setminus H)$ es el subespacio afín (si la interseccion es no vacia), y al completar $S=\widetilde{S} \setminus H$ recuperamos el $\widetilde{S}$.\\

Ahora veamos como pasar ecuaciones y coordenadas afines a proyectivas:

\[
c_0 + c_1 X_1 + \ldots + c_n X_n = 0 
\underbrace{\longrightarrow}_{\text{Homogeneización}}
c_0 x_0 + c_1 x_1 + \ldots + c_n x_n = 0
\]
\[
X_1 = \frac{x_1}{x_0}, X_2 = \frac{x_2}{x_0}, \ldots, X_n = \frac{x_n}{x_0}
\]
\[
c_0 + c_1 X_1 + \ldots + c_n X_n = 0 
\underbrace{\longleftarrow}_{\text{Deshomogeneización}}
c_0 x_0 + c_1 x_1 + \ldots + c_n x_n = 0
\]

Tiene sentido homogeneizar y deshomogeneizar, ya que cualquier ecuación polinómica: $1 + x_1 + x_2^2 = 0 \leftrightarrow x_0^2 + x_0 x_1 + x_2^2 = 0$ define el mismo conjunto de ceros en $\mathbb{P}^2$, salvo el punto $(0:0:1)$ que no pertenece a la carta afín.\\

A partir de ahora, a veces no utilizaremos notación distinta para coordenadas afines y proyectivas.

\begin{observación}
    Si $S,T \subset \mathbb{A}$ son subespacios afines, entonces son paralelas si:
    \begin{itemize}
        \item $S \cap T = \emptyset$
        \item $\vec{S} \subset \vec{T}$ o $\vec{S} \supset \vec{T}$
    \end{itemize}
    Al completar, tenemos que $\widetilde{S}, \widetilde{T} \subset \widetilde{\mathbb{A}}$ son subespacios proyectivos del completado proyectivo $\widetilde{\mathbb{A}}$ que cumplen:
    \[
    \widetilde{S} = S \cup S_{\infty} = S \cup \mathbb{P}(\vec{S}), \qquad
    \widetilde{T} = T \cup T_{\infty} = T \cup \mathbb{P}(\vec{T}),
    \]
    Se tiene que $\widetilde{S} \cap \widetilde{T} = S_{\infty} \cap T_{\infty} \subset \mathbb{A}_{\infty}$ siendo este el hiperplano de puntos en el infinito (siendo tambien $S$ y $T$ paralelos).\\
    En conclusion:
    \[
    S, T \text{paralelos} \iff
    S_{\infty} \subset T_{\infty} \text{ o } S_{\infty} \supset T_{\infty}  
    \]
    En particular, dos rectas afines son paralelas si y solo si sus completadas se cortan en un punto del hiperplano de puntos en el infinito.
\end{observación}

\begin{definición}[Aplicación afín]
    Una aplicación $f: \mathbb{A} \to \mathbb{A}'$ entre espacios afines es una \textbf{aplicación afín} si existe una aplicación lineal $\vec{f}: \vec{\mathbb{A}} \to \vec{\mathbb{A}}'$ tal que:
    \[
    \forall Q \in \mathbb{A}: \quad f(Q) = f(0) + \vec{f}(\overrightarrow{0Q})
    \]
\end{definición}

\begin{proposición}[Propiedades de las aplicaciones afines]
    Si $f: \mathbb{A} \to \mathbb{A}'$ es una aplicación afín, entonces:
    \begin{itemize}
        \item $f$ es inyectiva y/o sobreyectiva $\iff$ $\vec{f}$ es inyectiva y/o sobreyectiva.
        \item $f,g$ son aplicaciones afines $\implies$ $g \circ f$ es una aplicación afín, y $\vec{g \circ f} = \vec{g} \circ \vec{f}$.
        \item Si $S$ es un subespacio afín de $\mathbb{A}$, entonces $f(S)$ es un subespacio afín de $\mathbb{A}'$ y $\vec{f(S)} = \vec{f}(\vec{S})$.
        \item $f$ es afín y biyectiva $\implies$ $f^{-1}$ es afín y $\vec{f^{-1}} = \vec{f}^{-1}$.
    \end{itemize}
\end{proposición}

Veamos como escribir coordenadas afines.\\

Si $\mathfrak{R}_{\text{af}}$, $\mathfrak{R}_{\text{af}}'$ son referencias afines de $\mathbb{A}, \mathbb{A}'$ respectivamente, y $B, B'$ sus bases asociadas, con ademas $0$ el origen de $\mathfrak{R}_{\text{af}}$, entonces si $f(x_1, x_2, \ldots, x_n)_{\mathfrak{R}_{\text{af}}} = (y_1, y_2, \ldots, y_n)_{\mathfrak{R}_{\text{af}}'}$, tenemos que:
\[
    \begin{pmatrix}
        1 \\ y_1 \\ y_2 \\ \vdots \\ y_n
    \end{pmatrix} =
    \left(
    \begin{array}{c|ccc}
        1 & 0 & \cdots & 0 \\
        \hline
        q_1 & & & \\
        q_2 & & M_{B, B'}(\vec{f}) & \\
        \vdots & & & \\
        q_n & & & \\
    \end{array}
    \right)
    \begin{pmatrix}
        1 \\ x_1 \\ x_2 \\ \vdots \\ x_n
    \end{pmatrix}
\]

Siendo $f(0) = (q_1, q_2, \ldots, q_n)_{\mathfrak{R}_{\text{af}}'}$.

\begin{definición}[Completada proyectica de una aplicación afín]
    Sea $f: \mathbb{A} \to \mathbb{A}'$ una aplicación afín, la forma de extenderla a $\widetilde{f}: \widetilde{\mathbb{A}} \dashrightarrow \widetilde{\mathbb{A}}'$ es la siguiente:
    \[
    \begin{array}{ccc}
        \widetilde{\mathbb{A}} & \overset{\widetilde{f}}{\dashrightarrow} & \widetilde{\mathbb{A}}'  \\
        \| & & \| \\
        \mathbb{A} & \overset{f}{\rightarrow} & \mathbb{A}' \\
        \cup & & \cup \\
        \mathbb{A}_{\infty} = \mathbb{P}(\vec{\mathbb{A}}) & \overset{f_{\infty}}{\dashrightarrow} & \mathbb{A}_{\infty}' = \mathbb{P}(\vec{\mathbb{A}}') \\
    \end{array}
    \]
    Siendo $f_{\infty}$ la aplicación proyectiva asociada a la aplicación lineal $\vec{f}: \vec{\mathbb{A}} \to \vec{\mathbb{A}}'$.\\
\end{definición}

\begin{proposición}
    Si $\widetilde{f} : \mathbb{P} \dashrightarrow \mathbb{P}'$ es una aplicación proyectiva, y $\mathbb{H}, \mathbb{H}'$ son hiperplanos proyectivos de $\mathbb{P}, \mathbb{P}'$ respectivamente, tales que $\widetilde{f}(\mathbb{H}) \subset \mathbb{H}'$, y ademas $\widetilde{f}(\mathbb{P} \nsubseteq \mathbb{H}')$, entonces la restricción:
    \[
    f = \widetilde{f}_{|\mathbb{P} \setminus \mathbb{H}} : \mathbb{P} \setminus \mathbb{H} \to \mathbb{P}' \setminus \mathbb{H}'
    \]
    Es la restriccion de $\widetilde{f}$ a la carta afín $\mathbb{P} \setminus \mathbb{H}$, y es una aplicación afín cuya completada proyectiva es $\widetilde{f}$.
\end{proposición}

\begin{proposición}
    Son equivalentes:
    \begin{itemize}
        \item $f: \mathbb{A} \to \mathbb{A}'$ es una aplicación afín que envia rectas afines en rectas afines paralelas, es decir, $f(r) \parallel r$ con $r$ recta afín en $\mathbb{A}$.
        \item La completada proyectiva $\widetilde{f} : \widetilde{\mathbb{A}} \dashrightarrow \widetilde{\mathbb{A}}'$ es una aplicación proyectiva que fija el hiperplano de puntos en el infinito, es decir, $\widetilde{f}(\mathbb{A}_{\infty}) = \mathbb{A}_{\infty}'$.
    \end{itemize}
\end{proposición}


\subsection{Cálculo de puntos fijos e hiperplanos invariantes}

\begin{definición}
    Sea $f: \mathbb{P} \to \mathbb{P}$ una homografía.
    \begin{itemize}
        \item Un punto $A \in \mathbb{P}$ es un \textbf{punto fijo} de $f$ si $f(A) = A$.
        \item Un hiperplano $\mathbb{H} \subset \mathbb{P}$ es un \textbf{hiperplano invariante} de $f$ si $f(\mathbb{H}) = \mathbb{H}$.
        \item En general, un subespacio proyectivo $\mathbb{X} \subset \mathbb{P}$ es un \textbf{subespacio invariante} de $f$ si $f(\mathbb{X}) = \mathbb{X}$.
    \end{itemize}
\end{definición}

Fijamos un a referencia proyectiva $\mathfrak{R}_{\text{proy}}$, tengo la matriz $M_{\mathfrak{R}_{\text{proy}}}(f)$. Considerando $A = (x_0: x_1: \ldots : x_n)$ un punto fijo de $f$, tenemos que:
\[
M_{\mathfrak{R}_{\text{proy}}}(f) \begin{pmatrix}
    x_0 \\ x_1 \\ \vdots \\ x_n
\end{pmatrix} = \lambda \begin{pmatrix}
    x_0 \\ x_1 \\ \vdots \\ x_n
\end{pmatrix}, \quad \lambda \in \mathbb{K}^*
\]

Por tanto, $A=(x_0: x_1: \ldots : x_n)$ es un punto fijo de $f$ si y solo si es el autovector de algún autovalor $\lambda$ de $M_{\mathfrak{R}_{\text{proy}}}(f)$.

Denotamos 
\[
    \text{Fix}(f) = \{ A \in \mathbb{P} \mid f(A) = A \}
\]
Por cada $\lambda$ autovalor de $M_{\mathfrak{R}_{\text{proy}}}(f)$, considero su autoespacio
\[
    V_\lambda = \ker\{M_{\mathfrak{R}_{\text{proy}}}(f) - \lambda I\}
\]
Por lo general, acabamos de ver que
\[
    \text{Fix}(f) = \bigcup_{\lambda \in \text{Spec}(M_{\mathfrak{R}_{\text{proy}}}(f))} \mathbb{P}(V_\lambda)
\]
donde $\text{Spec}(M_{\mathfrak{R}_{\text{proy}}}(f))$ es el conjunto de autovalores de $M_{\mathfrak{R}_{\text{proy}}}(f)$.\\

\ejemplo{
    Sea $f: \mathbb{P}^2(\mathbb{R}) \circlearrowright $ con matriz en la referencia canónica:
    \[
    M_{\mathfrak{R}_{\text{proy}}}(f) = \begin{pmatrix}
        4 & -2 & 2 \\
        0 & 4 & 0 \\
        0 & -2 & 6
    \end{pmatrix}
    \]
    Entonces tenenmos los autovalores:
    \[
        \begin{vmatrix}
            4 - \lambda & -2 & 2 \\
            0 & 4 - \lambda & 0 \\
            0 & -2 & 6 - \lambda
        \end{vmatrix} = 0 \iff (4 - \lambda)^2 (6 - \lambda) = 0 \implies \lambda_1 = 2, \lambda_2 = 6
    \]
    con multiplicidades $m(\lambda_1) = 2$, $m(\lambda_2) = 1$.
    \[
        V_{\lambda=4} = \ker\begin{pmatrix}
            0 & -2 & 2 \\
            0 & 0 & 0 \\
            0 & -2 & 2
        \end{pmatrix}
    \]
    por lo tanto la base de $V_{\lambda=4}$ es $\{(1:0:0), (0:1:1)\}$, luego $\mathbb{P}(V_{\lambda=4})$ es una recta proyectiva generada por los puntos $(1:0:0), (0:1:1)$.
    \[
        V_{\lambda=6} = \ker\begin{pmatrix}
            -2 & -2 & 2 \\
            0 & -2 & 0 \\
            0 & -2 & 0
        \end{pmatrix} 
    \]
    por lo tanto la base de $V_{\lambda=6}$ es $\{(1, 0, 1)\}$, luego $\mathbb{P}(V_{\lambda=6})$ es el punto $(1:0:1)$.
}

\textbf{Cálculo de hiperplanos invariantes}\\

Todo hiperplano $H$ se puede escribir como:
\[
H: c_0 x_0 + c_1 x_1 + \ldots + c_n x_n = 0 \quad \text{coordenadas en referencia canónica}
\]
o que es lo mismo:
\[
    \begin{pmatrix}
        c_0 & c_1 & \cdots & c_n
    \end{pmatrix} \begin{pmatrix}
        x_0 \\ x_1 \\ \vdots \\ x_n
    \end{pmatrix} = 0
\]

Antes de ver cómo hallar los hiperplanos invariantes, observemos que si nos dan un hiperplano $H : c_0 x_0 + c_1 x_1 + \ldots + c_n x_n = 0$, y una homografía $f: \mathbb{P} \to \mathbb{P}$ con matriz $M_{\mathfrak{R}}(f)$, entonces ¿cuáles serán las ecuaciones de $f^{-1}(H)$?

\[
    (x_0: x_1: \ldots : x_n) \in f^{-1}(H) \iff f((x_0: x_1: \ldots : x_n)) \in H
\]
\[
    \iff (c_0, c_1, \ldots, c_n)_{\mathfrak{R}} \cdot \underbrace{M_{\mathfrak{R}}(f)}_{\text{coordenadas de $f(x_0 : \ldots : x_n)_{\mathfrak{R}}$}} \begin{pmatrix}
        x_0 \\ x_1 \\ \vdots \\ x_n
    \end{pmatrix} = 0 
\]
\[
    \iff \text{los coeficientes de las ecuaciones de } f^{-1}(H) \text{ son } (c_0', c_1', \ldots, c_n') = (c_0, c_1, \ldots, c_n) \cdot M_{\mathfrak{R}}(f)
\]
Si busco $H : c_0 x_0 + c_1 x_1 + \ldots + c_n x_n = 0$ hiperplano invariante, tenemos que:
\[
    f(H) = H \underbrace{\iff}_{\text{$f$ biyección}} f^{-1}(H) = H
\]
\[
    \iff \text{las ecuaciones los coeficientes $(c_0, c_1, \ldots, c_n)$ y $(c_0, c_1, \ldots, c_n) \cdot M_{\mathfrak{R}}(f)$ determinan el mismo hiperplano}
\] 
\[
    \iff \text{los coeficientes son proporcionales}
\]
Por otro lado, $H$ es un hiperplano invariante si y solo si existe $\mu \in \mathbb{K}^*$ tal que:
\[
    (c_0, c_1, \ldots, c_n) \cdot M_{\mathfrak{R}}(f) = \mu (c_0, c_1, \ldots, c_n)
\]
\[
    \underbrace{\iff}_{\text{trasponiendo}} M_{\mathfrak{R}}(f)^T \begin{pmatrix}
        c_0 \\ c_1 \\ \vdots \\ c_n
    \end{pmatrix} = \mu \begin{pmatrix}
        c_0 \\ c_1 \\ \vdots \\ c_n
    \end{pmatrix}
\]

\begin{observación}
    Hay una correlación entre los puntos fijos de $f$ y los hiperplanos invariantes de $f$ a través de la homografía dual $f^* : \mathbb{P}^* \to \mathbb{P}^*$. Pues
    \[
        M_{\mathfrak{R}^*}\begin{pmatrix}
            x_0 \\ x_1 \\ \vdots \\ x_n
        \end{pmatrix} = \lambda \begin{pmatrix}
            x_0 \\ x_1 \\ \vdots \\ x_n
        \end{pmatrix} \iff M_{\mathfrak{R}}(f)^T \begin{pmatrix}
            x_0 \\ x_1 \\ \vdots \\ x_n
        \end{pmatrix} = \lambda \begin{pmatrix}
            x_0 \\ x_1 \\ \vdots \\ x_n
        \end{pmatrix}
    \]
    entonces los autovalores y autovectores de $M_{\mathfrak{R}}(f)^T$ son los mismos que los de $M_{\mathfrak{R}}(f)$ y con la misma multiplicidad geométrica.
\end{observación}

\ejemplo{
    Volvemos al ejemplo anterior con la matriz:
    \[
    M_{\mathfrak{R}_{\text{proy}}}(f) = \begin{pmatrix}
        4 & -2 & 2 \\
        0 & 4 & 0 \\
        0 & -2 & 6
    \end{pmatrix}
    \]
    Que nos daba los autovalores $\lambda_1 = 4, \lambda_2 = 6$ con multiplicidades $m(\lambda_1) = 2, m(\lambda_2) = 1$.\\
    Si proyectivizamos obtenemos:
    \[
    \mathbb{P}(V_{\lambda=6}) = \{ (0:1:-1) \}
    \]
    \[
    \mathbb{P}(V_{\lambda=4}) = \text{recta proyectiva generada por } (1:0:0), (0:1:1) \equiv r: x_1 - x_2 = 0
    \]
    De lo que obtenemos los nucleos:
    \[
    \ker (M^t - 6I) = \ker \begin{pmatrix}
        -2 & 0 & 0 \\
        -2 & -2 & -2 \\
        2 & 0 & 0
    \end{pmatrix} = 
    \langle (0,1,-1) \rangle
    \]
    Obteniendo el hiperplano invariante:
    \[
    H : x_1 - x_2 = 0
    \]
    Ahora podemos obtener el otro nucleo:
    \[
    \ker (M^t - 4I) = \ker \begin{pmatrix}
        0 & 0 & 0 \\
        -2 & 0 & -2 \\
        2 & 0 & 2
    \end{pmatrix} = 
    \langle (1,0,0), (0,1,1) \rangle
    \]
    En realidad cualquier recta con coeficientes $\alpha (1,0,-1) + \beta (0,1,0)$, $\alpha, \beta \in \mathbb{R}$ sera invariante:
    \[
    H : \alpha x_0 + \beta x_1 - \alpha x_2 = 0, \quad \forall \alpha, \beta
    \]
    Teniendo así un haz de rectas de centro el punto $(1:0:1)$.
}

\begin{observación}
    Hay una correspondencia (dualidad) entre los puntos fijos de una homografía $f: \mathbb{P} \to \mathbb{P}$ y los hiperplanos invariantes de $f$, ya que:
    \begin{itemize}
        \item Los autovalores de $M_{\mathfrak{R}}(f)$ y $M_{\mathfrak{R}}(f)^T$ son los mismos.
        \item La dimensión del autoespacio asociado a cada autovalor es la misma tanto para $M_{\mathfrak{R}}(f)$ como para $M_{\mathfrak{R}}(f)^T$ (aunque los autoespacios en general difieren).
    \end{itemize}
    Esto se debe a que la trasposición de matrices mantiene determinantes y rangos respectivamente.
    \[
    \begin{array}{ccccccc}
        Fix(f) & = & \{Pto\} & \underbrace{\cup}_{+} & \{Pto\} & \underbrace{\cup}_{+} & recta \\
        & & \updownarrow & & \updownarrow & & \updownarrow \\
        & & H_{inv} & \underbrace{\cup}_{+} & H_{inv} & \underbrace{\cup}_{+} & haz\; de\; hiperplanos \\ 
    \end{array}
    \]
\end{observación}

\begin{observación}
    Si $X_1,X_2 \subset \mathbb{P}$ son subespacios proyectivos invariantes por una homografía $f: \mathbb{P} \to \mathbb{P}$, entonces su intersección $X_1 \cap X_2$ y su unión $X_1 + X_2$ también son subespacios proyectivos invariantes por $f$.
\end{observación}
