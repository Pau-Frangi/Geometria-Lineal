\section{Espacio proyectivo dual}

\begin{definición} [Espacio Vectorial Dual]
    Sea $V$ un espacio vectorial de dimensión $n+1$ sobre un cuerpo $\mathbb{K}$. El \textbf{espacio vectorial dual} de $V$, denotado por $V^*$, es el conjunto de todas las aplicaciones lineales de $V$ en $\mathbb{K}$. Es decir,
    \[
        V^* = \{ h : V \to \mathbb{K} \mid h \text{ es lineal} \}.
    \]
\end{definición}

Dada un base $\mathcal{B} = \{ \vec{u}_0, \vec{u}_1, \ldots, \vec{u}_n \}$ de $V$, existe una base dual $\mathcal{B}^* = \{ h_0, h_1, \ldots, h_n \}$ de $V^*$, donde cada $h_i$ es una forma lineal definida por:

\[
    \begin{matrix}
        h_i : V  \to  \mathbb{K} \\
        \vec{u}_j  \mapsto  \delta_{ij} =
        \begin{cases}
            1 & \text{si } i = j \\
            0 & \text{si } i \neq j
        \end{cases}
    \end{matrix}
\]

es base de $V^*$. Por lo tanto, $\dim(V^*) = n+1$. En este caso decimos que $\mathcal{B}^*$ es la \textbf{base dual} de $\mathcal{B}$.

\begin{definición} [Espacio Proyectivo Dual]
    En las condiciones anteriores, el \textbf{espacio proyectivo dual} de $\mathbb{P}=\mathbb{P}(V)$, denotado por $\mathbb{P}^*=\mathbb{P}(V^*)$, es el espacio proyectivo asociado a $V^*$. Es decir,
    \[
        \mathbb{P}^* = \mathbb{P}(V^*) = \{ [h] \mid h \in V^* \setminus \{0\} \}.
    \]
\end{definición}

Se tiene que $\dim(\mathbb{P}^*) = \dim(V^*) - 1 = \dim(\mathbb{P}) = n$.

\textbf{Interpretación:} Los puntos de $\mathbb{P}^*$ corresponden a los hiperplanos de $\mathbb{P}$. En efecto,

\[
    \underbrace{[h] \in \mathbb{P}^*}_{\text{punto}} \iff \underbrace{h \in V^* \setminus \{0\}}_{\text{recta vectorial}} \iff \underbrace{H = \{h = 0\} \subset V}_{\text{hiperplano vectorial}} \iff \underbrace{H \subset \mathbb{P}}_{\text{hiperplano proyectivo}}
\]

En coordenadas, 

\[
    \mathfrak{R} \text{ referencia en } \mathbb{P} \quad \Rightarrow \quad \mathcal{B} \text{ base de } V  
\]

\[
    \mathfrak{R}^* \text{ referencia en } \mathbb{P}^* \quad \Rightarrow \quad \mathcal{B}^* \text{ base dual de } V^*
\]

\[
    \mathbb{P}^* \ni [h] = [a_0 : a_1 : \ldots : a_n]_{\mathfrak{R}^*} \quad \Leftrightarrow \quad \mathbb{P} \supset H \equiv a_0x_0 + a_1x_1 + \ldots + a_nx_n = 0
\]
siendo $[x_0 : x_1 : \ldots : x_n]_{\mathfrak{R}}$ las coordenadas de un punto cualquiera de $\mathbb{P}$.

Dualidad entre los subespacios de $\mathbb{P}$ y $\mathbb{P}^*$:

\[
    \{ \text{Subespacios proyectivos de } \mathbb{P} \} \rightleftarrows  \{ \text{Subespacios proyectivos de } \mathbb{P}^* \}
\]
\[  
    X \to D(X) = \left\{[h] \in \mathbb{P}^* \mid X \subset H = \{h = 0\} \text{ (hiperplano proyectivo)} \right\}
\]
\[
    \Delta(X^*) = \bigcap_{[h] \in X^*} \{h = 0\} \subset \mathbb{P} \leftarrow X^* 
\]

$\Delta(X^*)$ es el \textbf{subespacio proyectivo} de $\mathbb{P}$ por ser intersección de hiperplanos proyectivos de $\mathbb{P}$.\\

$D(X)$ es \textbf{subespacio proyectivo} ya que si $X=P(W)$:\\
\[
    X \subset H \iff 
    W \subset \underbrace{\hat{H}}_{\text{hiperplano vectorial}} = \ker(h) \iff 
    h_{|W} = 0
\]

Por tanto:

\[
    X \subset H \iff
    h_{|W} = 0 \iff
    [h] \in \underbrace{Anu(W)}_{\text{anulador de } W} = \underbrace{\{ g \in V^* \mid g_{|W} = 0 \}}_{\text{subespacio vectorial de } V^*}
\]

Y decimos $D(X) = P(Anu(W))$.

\begin{lema}
    Sean $H, H_1, H_2, \ldots, H_k$ hiperplanos proyectivos de $\mathbb{P}$, entonces son equivalentes:
    \begin{enumerate}
        \item $H \supset H_1 \cap H_2 \cap \ldots \cap H_k$
        \item $H \in V \left( H_1, H_2, \ldots, H_k \right)$ en $\mathbb{P}^*$
        \item $[h] \in V \left( [h_1], [h_2], \ldots, [h_k] \right)$ en $\mathbb{P}^*$, donde $H_i = \{h_i = 0\} \quad \forall i = 1, 2, \ldots, k$
    \end{enumerate}
\end{lema}

\begin{proof}
    \[
    H \supset H_1 \cap H_2 \cap \ldots \cap H_k \text{ en } \mathbb{P} \iff
    \]
    \[
    \hat{H} \supset \hat{H}_1 \cap \hat{H}_2 \cap \ldots \cap \hat{H}_k \text{ en } V \iff
    \]
    \[
    h_1=0, h_2=0, \ldots, h_k=0 \implies h=0 \text{ en } V \iff
    \]
    \[
    h \in L \left( h_1, h_2, \ldots, h_k \right) \text{ en } V^* \iff
    \]
    \[
    [h] \in V\left( [h_1], [h_2], \ldots, [h_k] \right) \text{ en } \mathbb{P}^*
    \]
\end{proof}

Veamos la inyectividad de $\Delta$:

\begin{proof}
    Supongamos:
    \[
    \Delta(Y^*) \supset \Delta(X^*) \implies
    H \supset \Delta(X^*) \quad \forall H = \{h = 0\}: [h] \in Y^*
    \]
    Si ahora:
    \[
    X^* = V(H_0, H_1, \ldots, H_k) \underbrace{\implies}_{\text{Lema}}
    \Delta(X^*) = \bigcap_{i=0}^k H_i \implies
    \forall H \in Y^*: H \supset \Delta(X^*) = \bigcap_{i=0}^k H_i \iff
    \]
    \[
    \forall H \in Y^* \quad H \in V(H_0, H_1, \ldots, H_k) = X^* \implies
    Y^* \subset X^*
    \]
    Por tanto, finalmente:
    \[
    \Delta(Y^*) = \Delta(X^*) \implies
    \begin{cases}
        Y^* \subset X^* \\
        X^* \subset Y^*
    \end{cases} \implies
    X^* = Y^*
    \]
\end{proof}

Veamos la sobreyectividad de $D$:

\begin{proof}
    Dado $X^* = V(H_0, H_1, \ldots, H_r)$ subespacio proyectivo de $\mathbb{P}^*$, luego si definimos $X = \bigcap_{i=0}^r H_i$ tenemos que:
    \[
    D(X) =
    \{H \in P^* \mid X = \bigcap_{i=0}^r H_i \subset H \text{ en } \mathbb{P} \} \underbrace{=}_{\text{Lema}}
    V(H_0, H_1, \ldots, H_r) = X^* \text{ en } \mathbb{P}^*
    \] 
\end{proof}

Ademas, como $\Delta \circ D = \text{identidad}$, deducimos que $\Delta, D$ son biyecciones diccionario entre subespacios proyectivos del primal y dual.

\begin{observación}
    \textbf{Notación: } A partir de ahora, denotaremos $X^*=D(X)$ y diremos que $X^*$ es el subespacio proyectivo dual de $X$.\\
    Además, si $X$ es subespacio proyectivo, $X$ es intersección de hiperplanos proyectivos:
    \[
    \exists H_0, H_1, \ldots, H_k \text{ hiperplanos proyectivos tales que } X = \bigcap_{i=0}^k H_i
    \]
    Con ademas $\dim(X) = \underbrace{\dim(\mathbb{P})}_{n} - (k+1) = n - k - 1$.\\
    Entonces $X^* = V(H_0, H_1, \ldots, H_k)$ con $\dim(X^*) = k$, por tanto:
    \[
    \dim(X) + \dim(X^*) = n - k - 1 + k = n - 1
    \]
\end{observación}

\begin{proposición}[Propiedades fundamentales de la dualidad]
    Sea $\mathbb{P}$ un espacio proyectivo de dimensión $n$ y $X, Y \subset \mathbb{P}$ subespacios proyectivos. Entonces:
    \begin{enumerate}
        \item $X \subset Y \implies Y^* \subset X^*$
        \item $(X \cap Y)^* = V(X^*, Y^*)$
        \item $V(X, Y)^* = X^* \cap Y^*$
    \end{enumerate}
\end{proposición}
