\section{Espacios proyectivos}

\subsection{Definiciones y propiedades básicas}

\begin{definición}[Espacio proyectivo]
Dado un espacio vectorial $V$ se define el \textbf{espacio proyectivo} o el \textbf{proyectivizado} de $V$ y se denota por $\mathbb{P}(V)$ como el conjunto de rectas vectoriales de $V$ (esto es, los subespacios vectoriales de dimensión 1 de $V$).
A momentos nos olvidaremos de $V$ y llamaremos $\mathbb{P} = \mathbb{P}(V)$ espacio proyectivo.
\end{definición}

\begin{definición}[Dimension de un espacio proyectivo]
Si $V$ es un espacio vectorial, se define la \textbf{dimensión} del espacio proyectivo $\mathbb{P}(V)$ como:
\[    dim(\mathbb{P}(V)) = dim(V) - 1 \]
\end{definición}

\ejemplo{
    Si $V$ es un espacio vectorial de $dim=1$ (una recta vectorial) entonces todos los vectores de $V$ son proporcionales y por tanto $V$ solo tiene una recta vectorial. Por tanto $\mathbb{P}(V)$ es un punto y $dim(\mathbb{P}(V))=dim(V)-1=1-1=0$.
}

Los subespacios generados por un vector y por todos sus proporcionales son el mismo subespacio, por ejemplo:\\
$$ L[\vec{u}] = L[\vec{-2u}]$$

\ejemplo{
    Si $V = \{\vec{0}\}$ entonces $\mathbb{P}(V) = \emptyset$ y $dim(\mathbb{P}(V)) = dim(V) - 1 = 0 - 1 = -1$.
}

\textbf{Nota:} Pensaremos salvo que se diga lo contrario que el cuerpo de coeficientes de $V$ es $\mathbb{K} = \mathbb{R}$ ó $\mathbb{K} = \mathbb{C}$.\\

\begin{definición}[Espacio proyectivo canónico]
Se llama \textbf{espacio proyectivo canónico} sobre $\mathbb{K}$ de dimension $n$ al proyectivizado de $\mathbb{K}^{n+1}$.\\
Se denotaran por $\mathbb{P}^n(\mathbb{R})$ y $\mathbb{P}^n(\mathbb{C})$ respectivamente. Otras notaciones habituales son $\mathbb{R}\mathbb{P}^n$ y $\mathbb{C}\mathbb{P}^n$, ó $\mathbb{P}^n(\mathbb{R})$ y $\mathbb{P}^n(\mathbb{C})$.
\end{definición}

Alternativamente, podemos definir el espacio proyectivo como:
$$ \mathbb{P}(V) = \frac{V \setminus \{\vec{0}\}}{\sim} $$
Donde $\sim$ es la relación de equivalencia definida como:
$$ \vec{u} \sim \vec{v} \iff \exists \lambda \in \mathbb{K}^* : \vec{u} = \lambda \vec{v} $$
Es decir, dos vectores son equivalentes si son proporcionales.\\

\begin{definición}[Subespacio proyectivo]
Dado $\mathbb{P} = \mathbb{P}(V)$ un espacio proyectivo, llamaremos \textbf{subespacios proyectivos} (a veces subvariedades o variedades proyectivas) de $\mathbb{P}$ a los subconjuntos de la forma $\mathbb{P}(W)$ donde $W$ es un subespacio vectorial de $V$.
\end{definición}

\ejemplo{

    \begin{itemize}
        \item Pongamos de ejemplo los subespacios degenerados:\\
        $\{\vec{0}\}$, $V$ son subespacios vectoriales de $V$ y por tanto $\mathbb{P}(\{\vec{0}\}) = \emptyset$ y $\mathbb{P}(V) = \mathbb{P}$ son subespacios proyectivos de $\mathbb{P}$.
        \item Si $\mathbb{P}=\mathbb{P}(V)$ es un espacio proyectivo de dimensión 3 ($\dim(V)=4$).
    \end{itemize}

}



\observacion{
    A los subespacios proyectivos de codimension 1 (es decir, de $dim = dim(\mathbb{P}) - 1$) se les llama hiperplanos proyectivos.
}

\begin{lema}
    Dados dos puntos distintos $A$ y $B$ de un espacio proyectivo $\mathbb{P}$ existe una única recta que los contiene.
\end{lema}

\begin{proof}
    Sea $\mathbb{P} = \mathbb{P}(V)$, entonces $A = [\vec{u}]$ y $B = [\vec{v}]$ con $\vec{u}, \vec{v} \in V \setminus \{\vec{0}\}$.

    % Ejemplo visual: dos puntos proyectivos y la recta que los contiene
    \begin{center}
        \begin{tikzpicture}[scale=1.2]
            % Ejes coordenados
            \draw[->, gray] (0,0,0) -- (2,0,0) node[anchor=west] {$x$};
            \draw[->, gray] (0,0,0) -- (0,2,0) node[anchor=south] {$y$};
            \draw[->, gray] (0,0,0) -- (0,0,2) node[anchor=north east] {$z$};
            % Plano generado por u y v
            \filldraw[fill=blue!10, opacity=0.3] (0,0,0) -- (1.5,1.5,0) -- (1.5,0,1.5) -- cycle;
            % Vector u
            \draw[->, thick, blue] (0,0,0) -- (1,1,0) node[anchor=west] {$\vec{u}$};
            % Vector v
            \draw[->, thick, red] (0,0,0) -- (1,0,1) node[anchor=west] {$\vec{v}$};
            % Punto A
            \filldraw[blue] (1,1,0) circle (1.5pt) node[anchor=south west] {$A$};
            % Punto B
            \filldraw[red] (1,0,1) circle (1.5pt) node[anchor=north west] {$B$};
            % Recta proyectiva (intersección de plano y espacio)
            \draw[thick, purple, dashed] (0,0,0) -- (1.2,0.6,0.6);
        \end{tikzpicture}
    \end{center}

    Por ser $A \neq B$ tenemos que $\vec{u}$ y $\vec{v}$ no son proporcionales, luego son linealmente independientes. Así, tenemos que en $V$ el subespacio vectorial generado por $\vec{u}$ y $\vec{v}$ es:
    $$ W = L[\vec{u}, \vec{v}] \text{ con } dim(W) = 2 $$
    Por construcción $\vec{u}, \vec{v} \in W$ y por tanto $A, B \in \mathbb{P}(W)$. Dado que $dim(W) = 2$ tenemos que $\dim(\mathbb{P}(W)) = dim(W) - 1 = 1$, luego $\mathbb{P}(W)$ es una recta proyectiva.\\
    Además es única: si $W' \subset V$ es otro subespacio vectorial de dimensión 2 con $A, B \in \mathbb{P}(W')$ entonces $\vec{u}, \vec{v} \in W'$ y por tanto $W = L[\vec{u}, \vec{v}] \subseteq W'$. Como ambos tienen dimensión 2, $W = W'$ y por tanto $\mathbb{P}(W) = \mathbb{P}(W')$.
\end{proof}

\begin{proposición}[Propiedades de espacios proyectivos]
    Sea $\mathbb{P} = \mathbb{P}(V)$ un espacio proyectivo de dimensión $n$.
    \vspace{-1em}
    \begin{enumerate}
        \item $\mathbb{P}$ envia todos los subespacios vectoriales de $V$ en subespacios proyectivos de $\mathbb{P}$.
        \item $\mathbb{P}$ es sobreyectiva.
        \item $\mathbb{P}$ es inyectiva, es decir, $\mathbb{P}(W_1) = \mathbb{P}(W_2) \iff W_1 = W_2$.
        \item Si $W_1 \subseteq W_2 \implies \mathbb{P}(W_1) \subseteq \mathbb{P}(W_2)$.
    \end{enumerate}
\end{proposición}

\begin{proof}
    \leavevmode
    \begin{enumerate}
        \item Sean $W\subseteq V$ un subespacio vectorial. Por definición,
        \[
        \mathbb{P}(W)=\{[\vec w]\in\mathbb{P}(V)\ :\ \vec w\in W\setminus\{0\}\}.
        \]
        Si $W=\{0\}$ entonces $\mathbb{P}(W)=\emptyset$, que se considera el subespacio proyectivo vacío. Si $W\neq\{0\}$, sea $\dim W = r\ge 1$. Tomando una base $\{\vec w_1,\dots,\vec w_r\}$ de $W$ vemos que todo $[\vec w]\in\mathbb{P}(W)$ tiene un representante en el subespacio de dimensión $r$, por tanto $\mathbb{P}(W)$ es una variedad proyectiva de dimensión $r-1$. Es decir, $\mathbb{P}(W)$ es un subespacio proyectivo de $\mathbb{P}(V)$ y $\dim\mathbb{P}(W)=\dim W-1$.
        \item La aplicación $\pi:V\setminus\{0\}\to\mathbb{P}(V)$, $\pi(\vec v)=[\vec v]$, es por definición la proyección canónica que asigna a cada vector no nulo su clase de proporcionalidad. Dado cualquier punto $P\in\mathbb{P}(V)$ existe por definición algún $\vec v\in V\setminus\{0\}$ con $P=[\vec v]$, luego $\pi(\vec v)=P$. Por tanto $\pi$ es sobreyectiva.
        \item Supongamos primero que $W_1=W_2$. Entonces claramente $\mathbb{P}(W_1)=\mathbb{P}(W_2)$. Recíprocamente, supongamos $\mathbb{P}(W_1)=\mathbb{P}(W_2)$. Sea $\vec x\in W_1$. Si $\vec x=\vec0$ no hay nada que probar; si $\vec x\neq\vec0$ entonces $[\vec x]\in\mathbb{P}(W_1)=\mathbb{P}(W_2)$. Por definición de $\mathbb{P}(W_2)$ existe $\vec y\in W_2\setminus\{0\}$ y $\lambda\in K^\times$ tales que $\vec x=\lambda\vec y$ (porque representan la misma clase proyectiva). Entonces $\vec x\in W_2$. Hemos probado $W_1\subseteq W_2$. Intercambiando los papeles de $W_1$ y $W_2$ obtenemos la inclusión inversa, por lo que $W_1=W_2$.
        \item Si $W_1\subseteq W_2$ y $[\vec w]\in\mathbb{P}(W_1)$, entonces $\vec w\in W_1\setminus\{0\}\subseteq W_2\setminus\{0\}$, de modo que $[\vec w]\in\mathbb{P}(W_2)$. Por tanto $\mathbb{P}(W_1)\subseteq\mathbb{P}(W_2)$.
    \end{enumerate}
\end{proof}

\begin{proposición}[Intersección de subespacios proyectivos]
    Dados $\{X_i : i \in I\}$ subespacios proyectivos de $\mathbb{P} = \mathbb{P}(V)$, entonces su intersección $\bigcap_{i \in I} X_i$ es un subespacio proyectivo de $\mathbb{P}$.\\
\end{proposición}

\begin{proof}
    Tenemos que probar que $\bigcap_{i \in I} X_i = P(W)$ donde $W$ es un subespacio vectorial de $V$.\\
    Por hipótesis, $X_i = \mathbb{P}(W_i)$ donde $W_i$ es un subespacio vectorial de $V$.\\
    Veamos por tanto que $\bigcap_{i \in I} X_i = \mathbb{P}(\bigcap_{i \in I} W_i)$, siendo este ultimo un subespacio vectorial de $V$ al ser intersección de subespacios vectoriales.\\
    Cada elemento de $\bigcap_{i \in I} X_i$ es una recta vectorial que pertenece a todos los $X_i$, es decir, a todos los $\mathbb{P}(W_i)$, luego esta contenida en todos los $W_i$ y por tanto en $\bigcap_{i \in I} W_i$.\\
\end{proof}

\begin{definición}[Subespacio generado]
    Sea $S \subseteq \mathbb{P} = \mathbb{P}(V)$ un subconjunto, se define el \textbf{subespacio proyectivo generado} por $S$, y se denota por $V(S)$ como el menor subespacio proyectivo que contiene a $S$.\\
\end{definición}

Veamos que la definicion tiene sentido, es decir que $V(S)$ existe y es único, definiéndolo de la siguiente manera:
\[
V(S) = \bigcap_{\substack{X \subseteq \mathbb{P} \\ S \subseteq X}} X
\]
Este conjunto es no vacío porque $\mathbb{P}$ es un subespacio proyectivo contenido en $\mathbb{P}$ y contiene a $S$.\\
Evidentemente $S \subset \bigcap_{\substack{X \subseteq \mathbb{P} \\ S \subseteq X}} X$, y la intersección es un subespacio proyectivo por la proposición anterior, y si no fuera el menor estaríamos intersecando también el menor.\\
En el caso en que $S$ es la unión de subespacios proyectivos (puntos, rectas, planos, etc) $X_1, X_2, \ldots, X_k$, el subespacio generado se denota por $V(X_1, X_2, \ldots, X_k)$ ó $X_1 + X_2 + \ldots + X_k$.

\ejemplo{

    \[
    r = V(A,B) = A + B
    \]

\begin{proposición}[Formula de Grassmann para espacios proyectivos]
    Si $X_1, X_2$ son subespacios proyectivos de $\mathbb{P}$ entonces:
    \[
    \dim(X_1 \cap X_2) + \dim(V(X_1, X_2)) = \dim(X_1) + \dim(X_2)
    \]
\end{proposición}

\begin{proof}
    Por definición
    \[
    X_1\cap X_2=\mathbb{P}(W_1\cap W_2),\qquad V(X_1,X_2)=\mathbb{P}(L[W_1,W_2]),
    \]
    donde \(L[W_1,W_2]\) denota el subespacio lineal generado por \(W_1\) y \(W_2\).
    
    Aplicando la fórmula de Grassmann en el espacio vectorial \(V\) obtenemos
    \[
    \dim(W_1\cap W_2)+\dim(L[W_1,W_2])=\dim W_1+\dim W_2.
    \]
    
    Recordando la relación entre dimensiones vectorial y proyectiva, para un subespacio no nulo \(W\) se tiene
    \(\dim\mathbb{P}(W)=\dim W-1\), y además, si \(W=\{0\}\) entonces \(\mathbb{P}(W)=\emptyset\) y adoptamos \(\dim\emptyset=-1\). Restando \(1\) en ambos lados de la identidad vectorial anterior se obtiene
    \[
    (\dim(W_1\cap W_2)-1)+(\dim(L[W_1,W_2])-1)=(\dim W_1-1)+(\dim W_2-1).
    \]
    Pero cada término entre paréntesis es precisamente la dimensión proyectiva correspondiente:
    \[
    \dim\mathbb{P}(W_1\cap W_2)+\dim\mathbb{P}(L[W_1,W_2])
    =\dim\mathbb{P}(W_1)+\dim\mathbb{P}(W_2).
    \]
    Es decir,
    \[
    \dim(X_1\cap X_2)+\dim(V(X_1,X_2))=\dim(X_1)+\dim(X_2),
    \]
    como queríamos demostrar.
\end{proof}

\begin{corolario}
    En un plano proyectivo $\mathbb{P}$ todo par de rectas se corta.
\end{corolario}

\begin{proof}
    Sea \(\mathbb{P}\) un plano proyectivo (es decir, \(\dim\mathbb{P}=2\)) y sean
    \(r_1,r_2\) dos rectas proyectivas de \(\mathbb{P}\) (es decir, subespacios proyectivos de dimensión \(1\)).
    Aplicamos la fórmula de Grassmann para espacios proyectivos:
    \[
    \dim(r_1\cap r_2)+\dim(V(r_1,r_2))=\dim(r_1)+\dim(r_2).
    \]
    Como \(\dim(r_1)=\dim(r_2)=1\), la igualdad queda
    \[
    \dim(r_1\cap r_2)+\dim(V(r_1,r_2))=2.
    \]
    
    Por la definición de $V(r_1,r_2)$ sabemos que \(r_1\subseteq V(r_1,r_2)\) y \(r_2\subseteq V(r_1,r_2)\), luego
    \[
    1=\dim(r_i)\le \dim(V(r_1,r_2))\le \dim(\mathbb{P})=2,
    \]
    es decir \(\dim(V(r_1,r_2))\in\{1,2\}\).
        
    \begin{itemize}
        \item \textbf{Caso 1.} \(\dim(V(r_1,r_2))=1\).  
        Entonces \(V(r_1,r_2)\) es una recta que contiene a \(r_1\) y a \(r_2\). Por tanto \(r_1=r_2\). En este caso
        \(\dim(r_1\cap r_2)=\dim(r_1)=1\), es decir, la intersección es la recta completa (las rectas coinciden). 
        \item \textbf{Caso 2.} \(\dim(V(r_1,r_2))=2\).  
        Sustituyendo en la igualdad anterior:
        \[
        \dim(r_1\cap r_2)+2=2 \quad\Rightarrow\quad \dim(r_1\cap r_2)=0.
        \]
        Por tanto \(r_1\cap r_2\) es un punto (dimensión \(0\)).
    \end{itemize}   
\end{proof}
    

Nos podemos hacer la siguiente pregunta: ¿Que subespacios se pueden generar con $3$ puntos distintos $A_1, A_2, A_3$ de un espacio proyectivo $\mathbb{P}$?\\
Sabemos que cualquier par de puntos genera una recta $r=V(A_1, A_2) \underbracket{=}_{\text{notación}} A_1 + A_2$.\\
Entonces nos quedan dos casos:
\begin{itemize}
    \item $A_3 \in r$, luego $V(A_1, A_2, A_3) = r$, ya que $r=V(A_1, A_2) \subseteq V(A_1, A_2, A_3)$ claramente, y como $A_3 \in r \implies V(A_1, A_2, A_3) \subseteq r$.
    \item $A_3 \notin r$, luego $V(A_1, A_2, A_3)$ es un plano proyectivo ya que:
    \[
    \dim(\underbrace{V(A_1, A_2, A_3)}_{V(r, A_3)}) = \dim(r) + \dim(A_3) - \dim(r \cap A_3) = 1 + 0 - (-1) = 2
    \]
\end{itemize}

Conclusión: Tres puntos distintos de un espacio proyectivo $\mathbb{P}$ generan o bien una recta (si son alineados) o bien un plano (si no lo son).

\subsection{Independencia proyectiva}

\begin{lema}
    Sean $A_1 = [u_1], A_2 = [u_2], \ldots, A_r = [u_r]$ puntos distintos de un espacio proyectivo $\mathbb{P} = \mathbb{P}(V)$, entonces son equivalentes:
    \begin{itemize}
        \item[1)] Ningún punto pertenece al subespacio proyectivo generado por los otros, es decir:
        \[
        A_i \notin V(\underbrace{A_1, A_2, \ldots, \hat{A_i}, \ldots, A_r}_{\{A_1, A_2, \ldots, A_r\} \setminus \{A_i\}}) \quad \forall i=1,2,\ldots,r
        \]
        \item[2)] Los vectores $\{u_1, u_2, \ldots, u_r\}$ son linealmente independientes.
    \end{itemize}
\end{lema}

\begin{proof}
    Primero observamos que la 2ª condicion no depende de los representantes $u_i$ de los puntos $A_i$ escogidos ya que:
    \[
    \{u_1, u_2, \ldots, u_r\} \text{ son l.i.} \iff \{\lambda_1 u_1, \lambda_2 u_2, \ldots, \lambda_r u_r\} \text{ son l.i.} \quad \forall i \in \{1, \ldots, r\} \forall \lambda_i \in \mathbb{K}^* \setminus \{0\}
    \]
    Veamos ahora si la equivalencia:
    \[
    \{u_1, u_2, \ldots, u_r\} \text{ son l.i.} \iff
    \]
    \[
    \forall i \in \{1, \ldots, r\} : u_i \notin L[u_1, \ldots, \hat{u_i}, \ldots, u_r] \iff
    \]
    \[
    \forall i \in \{1, \ldots, r\} : [u_i] \cap L[u_1, \ldots, \hat{u_i}, \ldots, u_r] = \{0\} \underbrace{\iff}_{\text{pasando del vectorial al proyectivo}} 
    \]
    \[
    \forall i \in \{1, \ldots, r\} : P([u_i]) \cap P(L[u_1, \ldots, \hat{u_i}, \ldots, u_r]) = P(\{0\}) = \emptyset \iff
    \]
    \[
    \forall i \in \{1, \ldots, r\} : A_i \cap V(A_1, \ldots, \hat{A_i}, \ldots, A_r) = \emptyset \iff
    \]
    \[
    \forall i \in \{1, \ldots, r\} : A_i \notin V(A_1, \ldots, \hat{A_i}, \ldots, A_r)
    \]
    Y así estaria probada la equivalencia.
\end{proof}

\begin{definición}[Independencia proyectiva]
    Los puntos $A_1, A_2, \ldots, A_r$ de un espacio proyectivo $\mathbb{P}$ se dicen proyectivamente independientes si cumplen las condiciones del lema anterior, es decir, si ninguno de ellos pertenece al subespacio proyectivo generado por los otros, o equivalentemente, si los vectores que los representan son linealmente independientes.\\
    A veces tambien se dice que los puntos estan "en posición general".
\end{definición}

\begin{corolario}
    El subespacio proyectivo generado por $r$ puntos proyectivamente independientes tiene dimensión $r-1$.
\end{corolario}

\begin{proof}
    \[
    V(A_1, A_2, \ldots, A_r) = \mathbb{P}(\underbrace{L[u_1, u_2, \ldots, u_r]}_{\dim = r} \text{ al ser } \{u_1, \ldots, u_r\} \text{ l.i.}) \implies \dim(V(A_1, A_2, \ldots, A_r)) = r-1
    \]
\end{proof}

\ejemplo{
    Tres puntos $A_1, A_2, A_3$ son proyectivamente independientes si no estan alineados, es decir, si no estan en la misma recta proyectiva.\\
    En 5 puntos proyectivamente independientes no hay 3 alineados ni 4 coplanarios (que pertenezcan al mismo plano proyectivo).\\
    Tampoco pertenecen los 5 puntos a un subespacio de dimension 3.
}

\begin{lema}
    Dados $A_1, A_2, \ldots, A_r$ puntos de un espacio proyectivo. Son equivalentes:
    \begin{enumerate}
        \item $A_1, A_2, \ldots, A_r$ son proyectivamente independientes.
        \item $A_1, A_2, \ldots, A_r$ generan un subespacio proyectivo de dimension $r-1$.
    \end{enumerate}
\end{lema}

\begin{proof}
    Es el análogo proyectivo al enunciado vectorial.
    \[
        \{u_1, u_2, \ldots, u_r\} \text{ son l.i.} \iff \dim(L[u_1, u_2, \ldots, u_r]) = r
    \]
\end{proof}

\begin{corolario}
    En un espacio proyectivo de dimension $n$, los subconjuntos de puntos proyectivamente independientes tienen a lo sumo $n+1$ elementos.
\end{corolario}


\subsection{Coordenadas}

Consideramos el espacio proyectivo $\mathbb{P}^2_{\mathbb{R}} = \mathbb{P}(\mathbb{R}^3)$, es decir, el proyectivizado del espacio vectorial $\mathbb{R}^3$. ¿Cómo podemos dar coordenadas para $A \in \mathbb{P}^2_{\mathbb{R}}$?

Si $u = (u_0, u_1, u_2) \in \mathbb{R}^3 \setminus \{0\}$ es un vector con las coordenadas dadas en la base canónica de $\mathbb{R}^3$, entonces no podemos asignar a $A = [u]$ las coordenadas $(u_0, u_1, u_2)$ ya que si $v = \lambda u$ con $\lambda \in \mathbb{R}^* \setminus \{0\}$ entonces $[u] = [v]$ pero $(u_0, u_1, u_2) \neq (v_0, v_1, v_2)$.

Por ello, las \emph{coordenadas homogéneas} de un punto proyectivo
\(A \in \mathbb{P}^2_{\mathbb{R}}\) se escriben como
\[
A = (u_0 : u_1 : u_2),
\]
y se entienden definidas \emph{salvo un factor de proporcionalidad no nulo}.


\begin{definición} [Referencias proyectivas]
    Una \emph{referencia proyectiva} \(\mathfrak{R}\) de un espacio proyectivo 
    \(\mathbb{P}\) de dimensión \(n\) es un conjunto de \(n+2\) puntos
    \[
    \{A_0, A_1, \ldots, A_n, A_{n+1}\} 
    \]
    tales que cualesquiera \(n+1\) de ellos son proyectivamente independientes.\\
    Al punto \(A_{n+1}\) se le denomina \emph{punto unidad}.
\end{definición}

\ejemplo{
    \begin{itemize}
        \item Si $n=1$, $\mathbb{P} \equiv$ recta proyectiva, entonces cualesquiera tres puntos $A_0, A_1, A_2$ distintos forman una referencia proyectiva.
        \item Si $n=2$, $\mathbb{P} \equiv$ plano proyectivo, una referencia proyectiva la forman cuatro puntos $A_0, A_1, A_2, A_3$ tales que no hay tres alineados.
    \end{itemize}
}

\begin{proposición}
    Dada una referencia proyectiva $\mathfrak{R} = \{A_0, A_1, \ldots, A_n; A_{n+1}\}$ de un espacio proyectivo $\mathbb{P} = \mathbb{P}(V)$ de dimensión $n$, existe una única base salvo proporcionalidad $\mathcal{B} = \{u_0, u_1, \ldots, u_n\}$ de $V$ tal que:
    \[
    A_i = [u_i] \quad i=0,1,\ldots,n \quad \text{y} \quad A_{n+1} = [u_0 + u_1 + \ldots + u_n]
    \]
\end{proposición}

\begin{proof}
    Tomamos $v_i \in V$ tal que $A_i = [v_i]$ para $i=0,1,\ldots,n+1$.\\
    Como $\mathfrak{R}$ es una referencia proyectiva, $\{A_0, A_1, \ldots, A_n\}$ son proyectivamente independientes, luego $\{v_0, v_1, \ldots, v_n\}$ son linealmente independientes y por tanto forman una base de $V$.\\
    Por lo tanto, $v_{n+1} = \lambda_0 v_0 + \lambda_1 v_1 + \ldots + \lambda_n v_n$ con $\lambda_i \in \mathbb{R}$.\\
    Definimos $u_i = \lambda_i v_i$ para $i=0,1,\ldots,n$ y $u_{n+1} = v_{n+1}$, luego:
    \begin{itemize}
        \item $u_i \neq 0$ ya que $\lambda_i \neq 0$. En efecto, si $\lambda_i = 0$ entonces $A_{n+1} \in V(A_0, A_1, \ldots, \hat{A_i}, \ldots, A_n)$ lo que contradice la hipótesis de que $\mathfrak{R}$ es una referencia proyectiva, pues $\{A_0, A_1, \ldots, \hat{A_i}, \ldots, A_n, A_{n+1}\}$ serían proyectivamente dependientes.
        \item $\mathcal{B} = \{u_0, u_1, \ldots, u_n\}$ es una base de $V$ ya que $\mathcal{B}$ se obtiene de la base $\{v_0, v_1, \ldots, v_n\}$ multiplicando cada vector por un escalar no nulo.
    \end{itemize}
    Además $\mathcal{B}$ es única salvo proporcionalidad:
    \begin{itemize}
        \item Una vez escogido $v_{n+1}$ la base es única.
        \item Si $v'_{n+1} = \lambda v_{n+1}$ entonces la base que saldría sería $\mathcal{B}' = \{\lambda_0 \lambda v_0, \lambda_1 \lambda v_1, \ldots, \lambda_n \lambda v_n\}$ que es la misma base salvo proporcionalidad.
    \end{itemize}
\end{proof}

Se tiene que para $\mathbb{P} = \mathbb{P}(V)$ que

\[
    \text{Referencia proyectiva } \mathfrak{R} \leftrightarrow \text{Base de } V \text{ salvo proporcionalidad}
\]

salvo proporcionalidad.

\begin{definición} [Coordenadas homogéneas]
    Sea $\mathbb{P} = \mathbb{P}(V)$ un espacio proyectivo, y sea $\mathfrak{R} = \{A_0, A_1, \ldots, A_n; A_{n+1}\}$ una referencia proyectiva de $\mathbb{P}$ con base $\mathcal{B}$. Sea $A = [u] \in \mathbb{P}$ un punto proyectivo con $u = (x_0, x_1, \ldots, x_n)_\mathcal{B}$.\\
    Decimos que $(x_0 : x_1 : \ldots : x_n)$ son las \textbf{coordenadas homogéneas} de $A$ respecto a la referencia proyectiva $\mathfrak{R}$.
\end{definición}

\textbf{Nota:} Las coordenadas homogéneas se deben entender siempre salvo un factor de proporcionalidad no nulo. Por ejemplo:

\[
    (1 : 0 : -3)_\mathcal{B} = (2 : 0 : -6)_\mathcal{B} 
\]

Así, $\mathbb{P} \to \mathbb{K}^{n+1} \setminus \{0\} / \sim$ donde $\sim$ es la relación de equivalencia definida como:

\[
    (x_0, x_1, \ldots, x_n) \sim (y_0, y_1, \ldots, y_n) \iff \exists \lambda \in \mathbb{K}^* : (x_0, x_1, \ldots, x_n) = (\lambda y_0, \lambda y_1, \ldots, \lambda y_n)
\]

es decir, dos vectores son equivalentes si son proporcionales.

Por tanto, $(0, \ldots, 0)$ no son coordenadas homogéneas de ningún punto proyectivo (pues no hay orígen en un espacio proyectivo).

\ejemplo{
    Cogemos como espacio proyectivo a $\mathbb{P}^2_{\mathbb{R}} = \mathbb{P}(\mathbb{R}^3)$ y como referencia proyectiva 
    \[
        \mathfrak{R} = \{A_0=(1:0:-1), A_1=(3:0:-1), A_2=(1:-2:0); A_3=(0:2:1)\}
    \]
    Tomamos
    \[
        \begin{cases}
            v_0 = (1,0,-1) \\
            v_1 = (3,0,-1) \\
            v_2 = (1,-2,0) \\
            v_3 = (0,2,1)
        \end{cases} \text{ de forma que } [v_i] = A_i \quad i=0,1,2,3
    \]
    Entonces $\mathfrak{R}$ es una referencia proyectiva si y solo si cualesquiera $3$ vectores de $\{v_0, v_1, v_2, v_3\}$ son l.i.
    \[
        \iff \begin{pmatrix}
            1 & 3 & 1 & 0\\
            0 & 0 & -2 & 2\\
            -1 & -1 & 0 & 1
        \end{pmatrix} \text{ todos los menores } 3 \times 3 \text{ son }  \neq 0
    \]
    De forma alternativa, se puede definir si $\mathfrak{R}$ es referencia calculando su base asociada $\mathcal{B}$.
    \[
        v_3 = \lambda_0 v_0 + \lambda_1 v_1 + \lambda_2 v_2 
    \]
    \[
        \begin{cases}
            0 = \lambda_0 + 3\lambda_1 + \lambda_2 \\
            2 = -2\lambda_2 \\
            1 = -\lambda_0 - \lambda_1
        \end{cases} \implies \begin{cases}
            \lambda_2 = -1 \\
            \lambda_0 + 3\lambda_1 = 1 \\
            \lambda_0 + \lambda_1 = -1
        \end{cases} \implies \begin{cases}
            \lambda_2 = -1 \\
            \lambda_1 = 1 \\
            \lambda_0 = -2
        \end{cases}
    \]
    luego si 
    \[
        \begin{matrix}
            u_0 = \lambda_0 v_0 = -2(1,0,-1) = (-2,0,2) \\
            u_1 = \lambda_1 v_1 = 1(3,0,-1) = (3,0,-1) \\
            u_2 = \lambda_2 v_2 = -1(1,-2,0) = (-1,2,0) \\
            u_3 = v_3 
        \end{matrix}
    \]
    se cumple $u_3 = u_0 + u_1 + u_2 \implies$ base asociada $\mathcal{B} = \{u_0, u_1, u_2\}$. \\
    \textbf{Observación:} Si $\mathfrak{R}$ no fuera referencia, el sistema anterior sería incompatible o indeterminado.
}

\ejemplo{
    ¿Qué coordenadas tiene $C = (3:2:1)$ respecto a la referencia proyectiva $\mathfrak{R}$ del ejemplo anterior?
    \[
        (3:2:1) = [v = (3,2,1)]
    \]
    coordenadas de $v = (3,2,1)$ respecto a la base $\mathcal{B}$:
    \[
        v = \alpha_0 u_0 + \alpha_1 u_1 + \alpha_2 u_2 \implies \begin{cases}
            3 = -2\alpha_0 + 3\alpha_1 - \alpha_2 \\
            2 = 2\alpha_2 \\
            1 = 2\alpha_0 - \alpha_1
        \end{cases} \implies \begin{cases}
            4 = -2\alpha_0 + 3\alpha_1 \\
            1 = 2 \alpha_0 - \alpha_1 \\
            \alpha_2 = 1
        \end{cases} \implies \begin{cases}
            \alpha_0 = \frac{7}{4} \\
            \alpha_1 = \frac{5}{2} \\
            \alpha_2 = 1
        \end{cases}
    \]
    luego $C = (7:10:4)$ respecto a la referencia proyectiva $\mathfrak{R}$.
}

\begin{definición} [Referencia proyectiva canónica]
    La \textbf{referencia proyectiva canónica} de $\mathbb{P}^n_{\mathbb{K}} = \mathbb{P}(\mathbb{K}^{n+1})$ es la que tiene la base canónica de $\mathbb{K}^{n+1}$ como base asociada, es decir:
    \[
        \mathfrak{R}_{c} = \{A_0=(1:0:\ldots:0), A_1=(0:1:0:\ldots:0), \ldots, A_n=(0:\ldots:0:1); A_{n+1}=(1:1:\ldots:1)\}
    \]
\end{definición}

\textbf{Ecuaciones:}\\
En $\mathbb{P}^2_{\mathbb{R}}$ tenemos coordenadas $(x_0 : x_1 : x_2)$ homogéneas (por defecto se toma la referencia proyectiva canónica).

\begin{itemize}
    \item La ecuación $x_0 + x_1 - 2x_2 = 2$ no define un subconjunto de $\mathbb{P}^2_{\mathbb{R}}$ ya que si $(x_0 : x_1 : x_2)$ es solución, entonces $(2x_0 : 2x_1 : 2x_2)$ no lo es, sin embargo ambos representan el mismo punto proyectivo.
\end{itemize}

Necesitamos que las ecuaciones en coordenadas sean homogéneas, como por ejemplo:
\[
    x_0 + x_1 - 2x_2 = 0
\]

\begin{definición}[Cambio de referencia proyectiva]
    Sean $\mathfrak{R} = \{A_0, A_1, \ldots, A_n; A_{n+1}\}$ y $\mathfrak{R}' = \{A'_0, A'_1, \ldots, A'_n; A'_{n+1}\}$ dos referencias proyectivas de un espacio proyectivo $\mathbb{P}$, con bases asociadas $\mathcal{B} = \{u_0, u_1, \ldots, u_n\}$ y $\mathcal{B}' = \{u'_0, u'_1, \ldots, u'_n\}$ respectivamente.\\
    Si $A=\left(x_0 : x_1 : \ldots : x_n\right)$ y $A'=\left(x'_0 : x'_1 : \ldots : x'_n\right)$ son las coordenadas homogéneas de un punto $A \in \mathbb{P}$ respecto a las referencias $\mathfrak{R}$ y $\mathfrak{R}'$ respectivamente, entonces:
    \[
    \lambda
        \begin{pmatrix}
            \underset{..}{x'_0} \\
            \underset{..}{x'_1} \\
            \underset{..}{\vdots} \\
            x'_n
        \end{pmatrix} = M_{\mathcal{B}'\mathcal{B}} \cdot 
        \begin{pmatrix}
            \underset{..}{x_0} \\
            \underset{..}{x_1} \\
            \underset{..}{\vdots} \\
            x_n
        \end{pmatrix}
    \]
    Para algún $\lambda \in \mathbb{K}^* \setminus \{0\}$, donde $M_{\mathcal{B}'\mathcal{B}}$ es la matriz de cambio de base de $\mathcal{B}$ a $\mathcal{B}'$, y se llamara tambien matriz de cambio de referencia o de coordenadas.
\end{definición}
